# 2. メール本文レンダリング部分の修正

通常リンク（data-phished-url なし）のクリック処理を以下に変更：

const handleLinkClick = (e: MouseEvent) => {
  e.preventDefault()
  const target = e.currentTarget as HTMLAnchorElement
  const phishedUrl = target.dataset.phishedUrl

  if (phishedUrl) {
    // 罠メールのリンク → /phished に遷移
    tracker.onLinkClick(target.dataset.displayUrl ?? '')
    window.open(
      `${phishedUrl}&session_id=${sessionId}&email_id=${currentEmailId}&ui_lang=${uiLang}`,
      '_blank'
    )
  } else {
    // 通常メールのリンク → /safe に遷移
    window.open(
      `/safe?type=link&ui_lang=${uiLang}`,
      '_blank'
    )
  }
}
